{{- /* Image render hook. Add "#zoom" to an image destination to make it open enlarged in a lightbox on click, e.g. ![alt](schema.svg#zoom). */ -}} {{- $u := urls.Parse .Destination -}} {{- $zoom := eq $u.Fragment "zoom" -}} {{- $src := .Destination -}} {{- if not $u.IsAbs -}} {{- with or (.PageInner.Resources.Get $u.Path) (resources.Get $u.Path) -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery }}{{ $src = printf "%s?%s" $src . }}{{ end -}} {{- if and $u.Fragment (not $zoom) }}{{ $src = printf "%s#%s" $src $u.Fragment }}{{ end -}} {{- end -}} {{- end -}} {{- if and $zoom (eq $src .Destination) -}} {{- $src = strings.TrimSuffix "#zoom" $src -}} {{- end -}} {{- $img := printf `` $src (.PlainText | transform.HTMLEscape) (cond (ne .Title "") (printf ` title="%s"` (.Title | transform.HTMLEscape)) "") -}} {{- if $zoom -}} {{ $img | safeHTML }} {{- if not (.Page.Store.Get "imgZoomScript") -}} {{- .Page.Store.Set "imgZoomScript" true -}} {{- with resources.Get "js/img-zoom.js" | minify | fingerprint }} {{- end }} {{- end -}} {{- else -}} {{ $img | safeHTML }} {{- end -}}
